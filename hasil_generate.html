import html import json import time import requests import pandas as pd # ========================================== # TAHAP 1: EKSTRAKSI DATA (API BMKG -> DataFrame) # ========================================== # 1. Daftar kode wilayah (adm4): 4 kelurahan Kecamatan Purwokerto Utara, # Kabupaten Banyumas (kode Kemendagri 33.02.27.xxxx) daftar_kode_wilayah = { "33.02.27.1002": "Bancarkembar", "33.02.27.1005": "Grendeng", "33.02.27.1006": "Karangwangkal", "33.02.27.1004": "Pabuaran", } list_cuaca = [] print("Mengambil data dari beberapa lokasi...") # 2. Looping untuk setiap kode wilayah for kode, nama_cadangan in daftar_kode_wilayah.items(): url_api = f"https://api.bmkg.go.id/publik/prakiraan-cuaca?adm4={kode}" try: response = requests.get(url_api, timeout=10) except requests.RequestException as e: print(f"Gagal mengakses {kode}: {e}") continue if response.status_code == 200: data = response.json() # Pakai nama dari daftar di atas; beri peringatan jika berbeda dengan nama di API nama_desa = nama_cadangan desa_api = (data.get("lokasi") or {}).get("desa") if desa_api and desa_api.strip().lower() != nama_cadangan.lower(): print(f" [Perhatian] {kode}: nama di API '{desa_api}' != '{nama_cadangan}'") try: data_harian = data["data"][0]["cuaca"] for index_hari, prakiraan_harian in enumerate(data_harian, 1): for item in prakiraan_harian: # Masukkan ke list, tambahkan kolom Lokasi list_cuaca.append({ "Lokasi": nama_desa, "Hari_Ke": index_hari, "Waktu": item.get("local_datetime"), "Cuaca": item.get("weather_desc"), "Suhu_C": item.get("t"), "Kelembapan": item.get("hu") }) except (KeyError, IndexError): print(f"Struktur data tidak sesuai untuk {kode}, dilewati.") else: print(f"Gagal mengakses {kode} (status {response.status_code})") # Beri jeda agar tidak membebani server (etika scraping/API) time.sleep(2) # 3. Jadikan DataFrame df_cuaca = pd.DataFrame(list_cuaca) if df_cuaca.empty: raise SystemExit("Tidak ada data yang berhasil diambil. Periksa koneksi atau kode wilayah.") print(f"\nBerhasil! Total baris data sekarang: {len(df_cuaca)}") print(df_cuaca.head(10)) # 4. Simpan ke CSV (waktu tetap dalam format asli dari API) df_cuaca.to_csv("prakiraan_cuaca_bmkg_gabungan.csv", index=False) # ========================================== # TAHAP 2 & 3: GENERATE HTML & BOOTSTRAP 5 # ========================================== print("Memulai proses generate HTML...") # --- Persiapan kolom tampilan --- df_cuaca["No"] = df_cuaca.groupby("Lokasi", sort=False).cumcount() + 1 df_cuaca["Dt"] = pd.to_datetime(df_cuaca["Waktu"], errors="coerce") df_cuaca["Suhu_C"] = pd.to_numeric(df_cuaca["Suhu_C"], errors="coerce") df_cuaca["Kelembapan"] = pd.to_numeric(df_cuaca["Kelembapan"], errors="coerce") HARI = ["Sen", "Sel", "Rab", "Kam", "Jum", "Sab", "Min"] HARI_PANJANG = ["Senin", "Selasa", "Rabu", "Kamis", "Jumat", "Sabtu", "Minggu"] BULAN = ["Jan", "Feb", "Mar", "Apr", "Mei", "Jun", "Jul", "Agu", "Sep", "Okt", "Nov", "Des"] # Waktu sekarang dalam WIB (UTC+7), tanpa zona waktu agar bisa dibandingkan # dengan local_datetime BMKG. Aman dipakai di Colab (yang berjalan dalam UTC). SEKARANG_WIB = pd.Timestamp.now(tz="UTC").tz_localize(None) + pd.Timedelta(hours=7) waktu_buat = f"{SEKARANG_WIB.day} {BULAN[SEKARANG_WIB.month - 1]} {SEKARANG_WIB.year}, {SEKARANG_WIB:%H:%M}" def esc(teks): """Escape teks dari API sebelum dimasukkan ke HTML.""" return html.escape(str(teks)) def angka(nilai): return "-" if pd.isna(nilai) else str(int(round(nilai))) def jam_dari(dt): return 12 if pd.isna(dt) else dt.hour def format_waktu(dt, teks_asli): if pd.isna(dt): return teks_asli return f"{dt.day} {BULAN[dt.month - 1]} {dt.year}, {dt:%H:%M}" def ikon(deskripsi, jam): """Pilih emoji cuaca berdasarkan deskripsi BMKG dan jam (siang/malam).""" d = str(deskripsi or "").lower() malam = jam >= 18 or jam < 6 if "petir" in d: return "⛈️" if "hujan" in d: return "🌧️" if ("sedang" in d or "lebat" in d) else "🌦️" if "kabur" in d or "asap" in d or "kabut" in d: return "🌫️" if "cerah berawan" in d: return "☁️" if malam else "🌤️" if "berawan" in d: return "☁️" if "cerah" in d: return "🌙" if malam else "☀️" return "🌡️" df_cuaca["Waktu_Tampil"] = [ format_waktu(dt, w) for dt, w in zip(df_cuaca["Dt"], df_cuaca["Waktu"]) ] # --- Kerangka HTML + CDN Bootstrap 5 & Chart.js --- html_content = """ Data Cuaca BMKG - Kelompok 9
